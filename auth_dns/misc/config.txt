# --- Authoritative zones ---
# [name] header, then one record per line:
#   A / AAAA / NS / MX <prio> <host> / CNAME / TXT "..." ["..."] / SRV <p> <w> <port> <tgt>
#   HTTPS <prio> <target>
#   SOA <mname> <rname> <serial> <refresh> <retry> <expire> <minimum>
# Optional leading TTL on any record line, e.g. "300 A 192.168.1.2" (default 3600).
